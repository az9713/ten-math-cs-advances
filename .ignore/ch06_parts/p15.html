"""Validation subset for ch.6 (quantum parallel repetition). Needs numpy."""
import itertools
import math
import numpy as np


def report(name, ok, detail=""):
    print(f"{name}: {'OK' if ok else 'FAIL'} {detail}")


def vec(theta, out):
    c, s = math.cos(theta), math.sin(theta)
    return np.array([c, s]) if out == 0 else np.array([-s, c])


# 1. CHSH classical value = 3/4 (all 16 deterministic pairs)
best = max(sum((fa[x] ^ fb[y]) == (x & y)
               for x in (0, 1) for y in (0, 1)) / 4
           for fa in itertools.product((0, 1), repeat=2)
           for fb in itertools.product((0, 1), repeat=2))
report("chsh classical 3/4", abs(best - 0.75) < 1e-12, f"{best}")

# 2. CHSH rotation strategy on Phi+ = cos^2(pi/8)
tA, tB = (0.0, math.pi / 4), (math.pi / 8, -math.pi / 8)
w = sum(0.25 * (vec(tA[x], a) @ vec(tB[y], b)) ** 2 / 2
        for x in (0, 1) for y in (0, 1)
        for a in (0, 1) for b in (0, 1) if (a ^ b) == (x & y))
report("chsh quantum cos^2(pi/8)",
       abs(w - math.cos(math.pi / 8) ** 2) < 1e-12, f"{w:.6f}")


# 3. Tsirelson: |Bell operator| = 2 sqrt 2 at the optimal angles
def obs(t):
    v0, v1 = vec(t, 0), vec(t, 1)
    return np.outer(v0, v0) - np.outer(v1, v1)


A0, A1 = obs(0.0), obs(math.pi / 4)
B0, B1 = obs(math.pi / 8), obs(-math.pi / 8)
Bell = (np.kron(A0, B0) + np.kron(A0, B1)
        + np.kron(A1, B0) - np.kron(A1, B1))
nrm = np.linalg.norm(Bell, 2)
report("tsirelson 2 sqrt 2", abs(nrm - 2 * math.sqrt(2)) < 1e-12)


# 4. Feige game value = 1/2 (256 deterministic pairs)
def fwin(x, y, a, b):
    c, d = a // 2 + 1, a % 2
    return int(a == b and ((c == 1 and d == x) or (c == 2 and d == y)))


best = max(sum(fwin(x, y, fa[x], fb[y])
               for x in (0, 1) for y in (0, 1)) / 4
           for fa in itertools.product(range(4), repeat=2)
           for fb in itertools.product(range(4), repeat=2))
report("feige value 1/2", abs(best - 0.5) < 1e-12, f"{best}")

# 5. Feige cross strategy wins BOTH copies with prob 1/2
tot = sum(fwin(x1, y1, x1, y2) * fwin(x2, y2, 2 + x1, 2 + y2)
          for x1 in (0, 1) for x2 in (0, 1)
          for y1 in (0, 1) for y2 in (0, 1)) / 16
report("feige 2-fold cross 1/2", abs(tot - 0.5) < 1e-12, f"{tot}")

# 6. Pinsker on 300 random pairs
rng = np.random.default_rng(0)
ok = True
for _ in range(300):
    P = rng.random(4) + 1e-3
    Q = rng.random(4) + 1e-3
    P, Q = P / P.sum(), Q / Q.sum()
    dtv = 0.5 * np.abs(P - Q).sum()
    dkl = float((P * np.log(P / Q)).sum())
    ok &= dtv <= math.sqrt(0.5 * dkl) + 1e-12
report("pinsker 300 pairs", ok)

# 7. rejection-coupling disagreement = 2dTV/(1+dTV)
P = np.array([0.5, 0.3, 0.2])
Q = np.array([0.3, 0.3, 0.4])
d = 0.5 * np.abs(P - Q).sum()
exact = np.abs(P - Q).sum() / np.maximum(P, Q).sum()
report("corr-sampling formula",
       abs(exact - 2 * d / (1 + d)) < 1e-12, f"{exact:.4f}")

# 8. resolvent Born property: int (s/(s+u))^2 du = s
for sg in (0.3, 1.0, 2.7):
    u = np.linspace(0, 4000, 4_000_001)
    val = np.trapezoid((sg / (sg + u)) ** 2, u) + sg ** 2 / (sg + 4000)
    report(f"born integral s={sg}", abs(val - sg) < 1e-6, f"{val:.8f}")


# 9. Lemma 4.3 on a random 2x2 ensemble (exact resolvent Gram)
def gram(s, t):
    if abs(s - t) < 1e-12:
        return s
    return s * t * math.log(t / s) / (t - s)


def eigp(F):
    w, V = np.linalg.eigh(F)
    return [(w[k], V[:, k]) for k in range(len(w)) if w[k] > 1e-12]


def M(A, B):
    out = np.zeros(A.shape, complex)
    for s, a in eigp(A):
        for t, b in eigp(B):
            out += gram(s, t) * np.outer(a, a.conj()) @ np.outer(
                b, b.conj())
    return out


def H1(F):
    w, V = np.linalg.eigh(F)
    h = [-x * math.log(x) if x > 1e-15 else 0.0 for x in w]
    return (V * h) @ V.conj().T


def rc(d):
    H = rng.normal(size=(d, d)) + 1j * rng.normal(size=(d, d))
    H = H @ H.conj().T
    return H / np.linalg.eigvalsh(H).max()


ok = True
for _ in range(10):
    Fs = [rc(2) for _ in range(3)]
    ps = rng.random(3)
    ps /= ps.sum()
    Fb = sum(p * F for p, F in zip(ps, Fs))
    lhs = sum(p * (M(F, F) - M(F, Fb) - M(Fb, F) + M(Fb, Fb))
              for p, F in zip(ps, Fs))
    rhs = H1(Fb) - sum(p * H1(F) for p, F in zip(ps, Fs))
    ok &= np.linalg.eigvalsh(rhs - lhs).min() > -1e-8
report("lemma 4.3 entropy control", ok)

# 10. embezzlement error falls like 1/log b
errs = []
for k in (4, 8, 12, 16):
    b = 2 ** k
    j = np.arange(1, b + 1, dtype=float)
    c2 = (1 / j) / (1 / j).sum()
    lam = 0.3
    tgt = np.sort(np.concatenate([lam * c2, (1 - lam) * c2]))[::-1]
    src = np.sort(np.concatenate([c2, np.zeros(b)]))[::-1]
    F = np.sqrt(src * tgt).sum()
    errs.append(math.sqrt(max(0, 2 - 2 * F)))
report("embezzlement 1/log b",
       all(b < a for a, b in zip(errs, errs[1:])),
       " ".join(f"{e:.3f}" for e in errs))
